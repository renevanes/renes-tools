# Lokale AI-motor van Tools: llm.cpp bovenop llama.cpp (zie tools/llm-build.sh)
cmake_minimum_required(VERSION 3.16)
project(tools_llm CXX C)
set(CMAKE_CXX_STANDARD 17)
set(LLAMA_SRC "" CACHE PATH "map met de bron van llama.cpp")
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SRC} llama)
add_executable(llm llm.cpp)
target_include_directories(llm PRIVATE ${LLAMA_SRC}/vendor)
target_link_libraries(llm PRIVATE llama)
